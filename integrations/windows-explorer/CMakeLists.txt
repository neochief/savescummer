cmake_minimum_required(VERSION 3.21)
project(SaveScummerExplorer LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
    message(FATAL_ERROR "Explorer integration requires Windows and MSVC")
endif()
set(CMAKE_CXX_STANDARD 17)
get_filename_component(REPO "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
find_program(CARGO cargo HINTS "$ENV{USERPROFILE}/.cargo/bin" REQUIRED)
set(SAVESCUMMER_EXPLORER_DEV_DATA_DIR "" CACHE PATH "Isolated host data directory for the development extension")
if(SAVESCUMMER_EXPLORER_DEV_DATA_DIR)
    add_compile_definitions(SAVESCUMMER_EXPLORER_DEV)
endif()
set(BRIDGE "${CMAKE_CURRENT_BINARY_DIR}/cargo/release/savescummer_explorer_bridge.lib")
add_custom_target(explorer_bridge ALL COMMAND "${CMAKE_COMMAND}" -E env
    "SAVESCUMMER_EXPLORER_DEV_DATA_DIR=${SAVESCUMMER_EXPLORER_DEV_DATA_DIR}"
    "${CARGO}" build --locked --release -p savescummer-explorer-bridge --target-dir "${CMAKE_CURRENT_BINARY_DIR}/cargo"
    WORKING_DIRECTORY "${REPO}" BYPRODUCTS "${BRIDGE}")
add_library(savescummer-explorer SHARED shell.cpp shell.def)
add_dependencies(savescummer-explorer explorer_bridge)
target_link_libraries(savescummer-explorer PRIVATE "${BRIDGE}" shell32 ole32 uuid advapi32 userenv ws2_32 ntdll bcrypt propsys runtimeobject)
include(CTest)
if(BUILD_TESTING)
    add_executable(explorer-tests shell_test.cpp)
    target_link_libraries(explorer-tests PRIVATE ole32 uuid user32)
    add_dependencies(explorer-tests savescummer-explorer)
    add_test(NAME explorer-com COMMAND explorer-tests $<TARGET_FILE:savescummer-explorer>)
    # Invoked by dev-explorer.ps1 against its own isolated host and fresh saves.
    add_executable(explorer-smoke smoke_test.cpp)
    target_link_libraries(explorer-smoke PRIVATE shell32 ole32 uuid user32)
    add_dependencies(explorer-smoke savescummer-explorer)
endif()
